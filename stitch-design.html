async function calculateMisalDiscount(cart: Cart): Promise<number> {
  // Concurrent safe resolution with Promise.all
  const itemTotals = await Promise.all(
    cart.items.map(async (item) => {
      const res = await fetch(`/api/misal-rate/${item.id}`);
      const { data } = await res.json();
      return (data?.amount ?? 0) * item.quantity;
    })
  );

  const total = itemTotals.reduce((acc, curr) => acc + curr, 0);
  const discount = cart.user?.discount ?? 0;
  return Math.max(0, total - discount);
}
